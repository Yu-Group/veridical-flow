A full PCS pipeline in about 25 lines of sklearn-style code

from sklearn.model_selection import train_test_splitfrom sklearn.utils import resamplefrom sklearn.linear_model import LogisticRegressionfrom sklearn.tree import DecisionTreeClassifierfrom sklearn.metrics import accuracy_score, balanced_accuracy_scorefrom vflow import Vset, init_args X_train, X_test, y_train, y_test = init_args( train_test_split(X, y), names=["X_train", "X_test", "y_train", "y_test"]) subsampling = Vset(name="subsampling", vfuncs=[resample] * 3, output_matching=True)X_trains, y_trains = subsampling(X_train, y_train) modeling = Vset(name="modeling", vfuncs=[LogisticRegression(), DecisionTreeClassifier()], vfunc_keys=["LR", "DT"])modeling.fit(X_trains, y_trains)preds = modeling.predict(X_test) metrics = Vset(name="metrics", vfuncs=[accuracy_score, balanced_accuracy_score], vfunc_keys=["Acc", "Bal_Acc"])results = metrics.evaluate(preds, y_test)
